cmake_minimum_required(VERSION 3.16)
project(docking_openvins)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(docking_v6_msgs REQUIRED)
find_package(ov_msckf REQUIRED)
find_package(ov_core REQUIRED)
find_package(ov_init REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system thread)
find_package(Ceres REQUIRED)
add_executable(bridge src/bridge.cpp)
target_compile_features(bridge PRIVATE cxx_std_17)
target_compile_definitions(bridge PRIVATE ROS_AVAILABLE=2 ENABLE_ARUCO_TAGS=0)
target_include_directories(bridge PRIVATE ${EIGEN3_INCLUDE_DIRS} ${Boost_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${CERES_INCLUDE_DIRS})
ament_target_dependencies(bridge rclcpp docking_v6_msgs ov_msckf ov_core ov_init)
target_link_libraries(bridge ${OpenCV_LIBS} ${Boost_LIBRARIES} ${CERES_LIBRARIES})
install(TARGETS bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
